## 最佳快速上手方案（1 小时 Demo）

### 🎯 方案概述

使用 **MediaPipe** 进行手部关键点检测，结合 **FAISS** 进行向量相似度搜索，实现免训练的手语识别。

### 📊 技术架构

```
┌─────────────────────────────────────────────────────────────────┐
│                        摄像头实时输入                            │
└─────────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────────┐
│              MediaPipe Holistic 手部关键点检测                   │
│   • 21 个手部 landmarks                                          │
│   • 3D 坐标 (x, y, z)                                            │
│   • 实时检测 (30+ FPS)                                           │
└─────────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────────┐
│                    特征提取与归一化                              │
│   • 42 维特征向量 (21 点 × 2 坐标)                                 │
│   • 归一化处理 (0-1)                                             │
│   • 旋转/缩放不变性                                              │
└─────────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────────┐
│                  FAISS 向量相似度搜索                            │
│   • L2 距离计算                                                  │
│   • K 近邻搜索 (K=5)                                              │
│   • 毫秒级检索                                                   │
└─────────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────────┐
│                    结果输出与可视化                              │
│   • 手势名称显示                                                 │
│   • 置信度评分                                                   │
│   • 实时视频叠加                                                 │
└─────────────────────────────────────────────────────────────────┘
```

## 最简技术栈对比

### 方案 A：MediaPipe + FAISS（⭐ 强烈推荐）

```bash
# 核心依赖（仅 4 个）
pip install mediapipe opencv-python faiss-cpu numpy
```

| 包 | 版本要求 | 作用 |
|----|----------|------|
| `mediapipe` | >=0.10.0 | 手部关键点检测 |
| `opencv-python` | >=4.8.0 | 摄像头读取、图像处理 |
| `faiss-cpu` | >=1.7.4 | 向量相似度搜索 |
| `numpy` | >=1.23.0 | 数值计算 |

**优点**:
- ✅ 无需训练，立即可用
- ✅ 代码量 < 200 行
- ✅ 添加新手势简单（录制即可）
- ✅ 推理速度快（实时 30+ FPS）
- ✅ 内存占用小

**缺点**:
- ❌ 手势数量有限（依赖模板库大小）
- ❌ 对相似手势区分能力有限
- ❌ 不支持连续手语

**适用场景**: 快速 Demo、原型验证、小规模手势识别

### 📄 文件 2: `record_templates.py`（手势模板录制）

```python
"""
手势模板录制脚本
================
功能：通过摄像头录制手势，保存关键点特征作为模板

使用方法:
    python record_templates.py
    
操作说明:
    - 按空格键：保存当前手势帧
    - 按 'n' 键：切换到下一个手势
    - 按 'q' 键：退出程序
"""

import cv2
import mediapipe as mp
import numpy as np
import json
import os

# 创建模板目录
TEMPLATE_DIR = "templates"
os.makedirs(TEMPLATE_DIR, exist_ok=True)

# 初始化 MediaPipe Hands
mp_hands = mp.solutions.hands
mp_drawing = mp.solutions.drawing_utils

# 定义要录制的手势列表（支持中文）
GESTURES = ["hello", "ok", "peace", "fist", "thumbs_up"]

def extract_landmark_features(landmarks, image_width, image_height):
    """
    从手部关键点提取特征向量
    
    参数:
        landmarks: MediaPipe 手部关键点
        image_width: 图像宽度
        image_height: 图像高度
    
    返回:
        42 维特征向量 (21 个点的 x,y 坐标，归一化)
    """
    features = []
    for landmark in landmarks.landmark:
        # 归一化坐标 (0-1)
        x = landmark.x
        y = landmark.y
        features.extend([x, y])
    return np.array(features, dtype=np.float32)

def main():
    """主函数：录制手势模板"""
    cap = cv2.VideoCapture(0)
    
    # 存储所有手势数据
    all_gesture_data = {gesture: [] for gesture in GESTURES}
    
    current_gesture_idx = 0
    print(f"开始录制手势：{GESTURES[current_gesture_idx]}")
    print("按空格键保存帧，按'n'切换手势，按'q'退出")
    
    with mp_hands.Hands(
        static_image_mode=False,
        max_num_hands=1,
        min_detection_confidence=0.7,
```

### 📄 文件 3: `build_faiss_index.py`（FAISS 索引构建）

```python
"""
构建 FAISS 向量索引
==================
功能：将录制的手势模板转换为 FAISS 索引，用于快速相似度搜索

输入:
    templates/gesture_data.npy - 手势特征数据
    templates/gesture_labels.json - 手势标签

输出:
    templates/faiss_index.bin - FAISS 索引文件
    templates/label_map.json - 标签映射
"""

import faiss
import numpy as np
import json
import os

TEMPLATE_DIR = "templates"

def main():
    """构建 FAISS 索引"""
    # 加载手势数据
    print("加载手势数据...")
    gesture_data = np.load(
        os.path.join(TEMPLATE_DIR, "gesture_data.npy"),
        allow_pickle=True
    ).item()
    
    with open(os.path.join(TEMPLATE_DIR, "gesture_labels.json"), 'r') as f:
        gestures = json.load(f)
    
    # 准备训练数据
    all_vectors = []
    labels = []
    
    for i, gesture in enumerate(gestures):
        if gesture in gesture_data:
            vectors = np.array(gesture_data[gesture], dtype=np.float32)
            all_vectors.append(vectors)
            labels.extend([i] * len(vectors))
            print(f"  {gesture}: {len(vectors)} 个样本")
    
    if len(all_vectors) == 0:
        print("错误：没有找到手势数据！请先运行 record_templates.py")
        return
    
    # 合并所有向量
    X = np.vstack(all_vectors)
    labels = np.array(labels, dtype=np.int32)
    
    print(f"\n总样本数：{len(X)}")
    print(f"特征维度：{X.shape[1]}")
    print(f"手势类别：{len(gestures)}")
    
    # 构建 FAISS 索引（使用 L2 距离）
    print("\n构建 FAISS 索引...")
    dimension = X.shape[1]
    
    # 使用精确搜索（小数据集适用）
    index = faiss.IndexFlatL2(dimension)
    
    # 添加向量到索引
    index.add(X)
    
    print(f"索引包含 {index.ntotal} 个向量")
    
    # 保存索引
    index_path = os.path.join(TEMPLATE_DIR, "faiss_index.bin")
    faiss.write_index(index, index_path)
    print(f"索引已保存到：{index_path}")
    
    # 保存标签映射
    label_map = {i: gesture for i, gesture in enumerate(gestures)}
    with open(os.path.join(TEMPLATE_DIR, "label_map.json"), 'w') as f:
        json.dump(label_map, f)
    
    print("\n✅ FAISS 索引构建完成！")
    print("下一步：运行 real_time_recognition.py 进行实时识别")

if __name__ == "__main__":
    main()
```

### 📄 文件 4: `real_time_recognition.py`（实时识别）

```python
"""
实时手语识别
============
功能：使用 MediaPipe 检测手部，FAISS 匹配手势模板

输入:
    templates/faiss_index.bin - FAISS 索引
    templates/label_map.json - 标签映射

输出:
    实时视频显示识别结果
"""

import cv2
import mediapipe as mp
import faiss
import numpy as np
import json
import os

TEMPLATE_DIR = "templates"

# 初始化 MediaPipe Hands
mp_hands = mp.solutions.hands
mp_drawing = mp.solutions.drawing_utils

def extract_landmark_features(landmarks, image_width, image_height):
    """提取手部关键点特征"""
    features = []
    for landmark in landmarks.landmark:
        x = landmark.x
        y = landmark.y
        features.extend([x, y])
    return np.array(features, dtype=np.float32).reshape(1, -1)

def load_faiss_index():
    """加载 FAISS 索引和标签"""
    index_path = os.path.join(TEMPLATE_DIR, "faiss_index.bin")
    label_map_path = os.path.join(TEMPLATE_DIR, "label_map.json")
    
    if not os.path.exists(index_path):
        raise FileNotFoundError(
            "FAISS 索引不存在！请先运行 build_faiss_index.py"
        )
    
    index = faiss.read_index(index_path)
    
    with open(label_map_path, 'r') as f:
        label_map = {int(k): v for k, v in json.load(f).items()}
    
    return index, label_map

def main():
    """主函数：实时手语识别"""
    # 加载 FAISS 索引
    print("加载 FAISS 索引...")
    index, label_map = load_faiss_index()
    print(f"索引包含 {index.ntotal} 个向量")
    print(f"支持 {len(label_map)} 种手势")
    
    # 打开摄像头
    cap = cv2.VideoCapture(0)
    
    # 置信度阈值
    CONFIDENCE_THRESHOLD = 0.5
    # 搜索最近邻数量
    K_NEIGHBORS = 5
    
    print("\n开始实时识别...")
    print("按 'q' 键退出")
    
    with mp_hands.Hands(
        static_image_mode=False,
        max_num_hands=1,
        min_detection_confidence=0.7,
        min_tracking_confidence=0.5
    ) as hands:
        
        while cap.isOpened():
            ret, frame = cap.read()
            if not ret:
                break
            
            # 翻转图像（镜像）
            frame = cv2.flip(frame, 1)
            image = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            image.flags.writeable = False
            
            # 检测手部
            results = hands.process(image)
            
            image.flags.writeable = True
            image = cv2.cvtColor(image, cv2.COLOR_RGB2BGR)
            
            prediction_text = "No hand detected"
            confidence = 0.0
            
            if results.multi_hand_landmarks:
                for hand_landmarks in results.multi_hand_landmarks:
                    # 绘制关键点
                    mp_drawing.draw_landmarks(
                        image, hand_landmarks, mp_hands.HAND_CONNECTIONS)
                    
                    # 提取特征
                    features = extract_landmark_features(
                        hand_landmarks, frame.shape[1], frame.shape[0])
                    
                    # FAISS 搜索
                    distances, indices = index.search(features, K_NEIGHBORS)
                    
                    # 投票决定最终预测
                    predictions = []
                    for i, idx in enumerate(indices[0]):
                        if idx < len(label_map):
                            gesture = label_map[idx]
                            # 将距离转换为置信度（简单方法）
                            conf = 1.0 / (1.0 + distances[0][i])
                            predictions.append((gesture, conf))
                    
                    # 选择最高置信度的预测
                    if predictions:
                        predictions.sort(key=lambda x: x[1], reverse=True)
                        prediction_text = predictions[0][0]
                        confidence = predictions[0][1]
            
            # 显示结果
            status_color = (0, 255, 0) if confidence > CONFIDENCE_THRESHOLD else (0, 128, 255)
            cv2.putText(image, f"Prediction: {prediction_text}", 
                       (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 
                       1, status_color, 2)
            cv2.putText(image, f"Confidence: {confidence:.2f}", 
                       (10, 70), cv2.FONT_HERSHEY_SIMPLEX, 
                       0.7, (255, 255, 255), 2)
            
            cv2.imshow('Sign Language Recognition', image)
            
            if cv2.waitKey(1) & 0xFF == ord('q'):
                break
    
    cap.release()
    cv2.destroyAllWindows()
    print("\n识别结束")

if __name__ == "__main__":
    try:
        main()
    except FileNotFoundError as e:
        print(f"错误：{e}")
        print("请确保已运行 record_templates.py 和 build_faiss_index.py")
```

### ❓ Q3: 识别结果闪烁

**现象**: 预测结果在不同手势间快速切换

**解决方案**: 添加时序平滑

```python
# 方法 1: 简单投票平滑
from collections import Counter

class PredictionSmoother:
    def __init__(self, history_size=5):
        self.history = []
        self.history_size = history_size
    
    def smooth(self, current_pred):
        self.history.append(current_pred)
        if len(self.history) > self.history_size:
            self.history.pop(0)
        # 返回出现频率最高的预测
        return Counter(self.history).most_common(1)[0][0]

# 使用示例
smoother = PredictionSmoother(history_size=5)
smoothed_pred = smoother.smooth(current_prediction)

# 方法 2: 置信度加权平滑
class WeightedSmoother:
    def __init__(self, alpha=0.7):
        self.alpha = alpha  # 当前帧权重
        self.prev_pred = None
        self.prev_conf = 0
    
    def smooth(self, current_pred, current_conf):
        if self.prev_pred is None:
            self.prev_pred = current_pred
            self.prev_conf = current_conf
            return current_pred
        
        # 加权平均
        if current_conf > self.prev_conf:
            result = current_pred
        else:
            result = self.prev_pred
        
        self.prev_pred = result
        self.prev_conf = current_conf
        return result
```
