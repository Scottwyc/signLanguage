# 手语动作准确度评测系统技术方案

## 基于 MediaPipe + 特征比对的评分系统

**生成日期**: 2026-04-02  
**应用场景**: 手语学习/考试/训练场景，评测用户动作与标准动作的相似度

---

## 1. 需求分析

### 1.1 核心需求

| 需求 | 说明 | 技术要点 |
|------|------|---------|
| **标准动作库** | 存储标准手语示范视频/特征 | 每个语义至少 3-5 个标准样本 |
| **实时采集** | 摄像头拍摄用户动作 | 低延迟，实时反馈 |
| **特征比对** | 用户特征 vs 标准特征 | 逐点/逐帧相似度计算 |
| **准确度评分** | 给出 0-100 分或等级 | 多维度评分（手势、时序、流畅度） |
| **错误定位** | 指出具体哪里不准确 | 关键帧/关键关节差异分析 |
| **孤立词场景** | 单个手语词评测 | 简化时序建模 |

### 1.2 与原技术路线的差异

| 原方案（识别） | 新方案（评测） | 差异说明 |
|--------------|--------------|---------|
| 目标：识别是哪个手语词 | 目标：评估动作准确度 | 评测需要更细粒度比对 |
| 输出：语义类别 + 置信度 | 输出：分数 + 错误分析 | 需要可解释性 |
| 比对：特征向量整体相似度 | 比对：逐关节/逐帧差异 | 需要局部分析 |
| 容错：允许一定偏差 | 敏感：需要检测偏差 | 评测需要更敏感 |

## 3. 推荐技术方案

### 3.1 特征比对方法

#### 方法 1：整体相似度（快速粗评）

```python
import faiss
import numpy as np

def overall_similarity(user_feature, standard_features):
    """
    整体相似度计算
    user_feature: (D,) 用户特征向量
    standard_features: (N, D) N 个标准样本
    返回：0-1 相似度分数
    """
    # 归一化
    user_norm = user_feature / np.linalg.norm(user_feature)
    std_norms = standard_features / np.linalg.norm(standard_features, axis=1, keepdims=True)
    
    # 余弦相似度
    similarities = np.dot(std_norms, user_norm)
    
    # 返回最高相似度
    return np.max(similarities)
```

**特点**：快速，适合初筛，但无法定位具体差异

#### 方法 2：逐关节差异分析（精确定位）

```python
def joint_level_analysis(user_landmarks, standard_landmarks):
    """
    逐关节差异分析
    user_landmarks: (F, 543, 3) 用户特征 (帧数，关键点，坐标)
    standard_landmarks: (F, 543, 3) 标准特征
    返回：各部位差异分数
    """
    # 计算每个关键点的平均欧氏距离
    diff_per_point = np.mean(np.linalg.norm(
        user_landmarks - standard_landmarks, axis=2  # 3D 距离
    ), axis=0)  # 平均所有帧 → (543,)
    
    # 按部位分组
    left_hand = diff_per_point[:63].mean()    # 左手 21 点
    right_hand = diff_per_point[63:126].mean()  # 右手 21 点
    body = diff_per_point[126:225].mean()     # 身体 33 点
    face = diff_per_point[225:].mean()        # 面部
    
    return {
        'left_hand': left_hand,
        'right_hand': right_hand,
        'body': body,
        'face': face,
        'overall': np.mean([left_hand, right_hand, body, face])
    }
```

**特点**：可定位具体哪个部位不准确

#### 方法 3：DTW 时序对齐（处理速度差异）

```python
from scipy.spatial.distance import cdist
import dtw  # pip install dtw-python

def dtw_alignment(user_seq, standard_seq):
    """
    DTW 动态时间规整
    处理用户动作速度与标准不一致的情况
    user_seq: (T1, D) 用户特征序列
    standard_seq: (T2, D) 标准特征序列
    返回：对齐后的距离
    """
    # 计算距离矩阵
    dist_matrix = cdist(user_seq, standard_seq, metric='euclidean')
    
    # DTW 对齐
    alignment = dtw.dtw(dist_matrix)
    
    return alignment.distance, alignment
```

**特点**：处理动作快慢不一致，但计算较慢

### 3.2 评分模型

#### 综合评分公式

```python
def calculate_score(joint_diff, temporal_diff, speed_diff, weights=None):
    """
    综合评分（0-100 分）
    
    参数：
    - joint_diff: 关节差异 (0-1, 越小越好)
    - temporal_diff: 时序差异 (0-1, 越小越好)
    - speed_diff: 速度差异 (0-1, 越小越好)
    - weights: 各维度权重
    
    返回：0-100 分数
    """
    if weights is None:
        weights = {
            'joint': 0.5,      # 关节准确度 50%
            'temporal': 0.3,   # 时序准确度 30%
            'speed': 0.2       # 速度准确度 20%
        }
    
    # 转换为分数（差异越小分数越高）
    joint_score = (1 - joint_diff) * 100
    temporal_score = (1 - temporal_diff) * 100
    speed_score = (1 - speed_diff) * 100
    
    # 加权平均
    total_score = (
        joint_score * weights['joint'] +
        temporal_score * weights['temporal'] +
        speed_score * weights['speed']
    )
    
    return {
        'total': round(total_score, 1),
        'joint': round(joint_score, 1),
        'temporal': round(temporal_score, 1),
        'speed': round(speed_score, 1),
        'grade': get_grade(total_score)
    }

def get_grade(score):
    """分数转等级"""
    if score >= 90:
        return '优秀'
    elif score >= 80:
        return '良好'
    elif score >= 60:
        return '及格'
    else:
        return '不及格'
```

### 3.3 错误分析报告

```python
def generate_feedback(user_landmarks, standard_landmarks, score_result):
    """
    生成错误反馈报告
    """
    feedback = {
        'score': score_result['total'],
        'grade': score_result['grade'],
        'strengths': [],
        'weaknesses': [],
        'suggestions': []
    }
    
    # 分析各部位差异
    joint_analysis = joint_level_analysis(user_landmarks, standard_landmarks)
    
    # 找出最好的和最差的部位
    best_part = min(joint_analysis, key=joint_analysis.get)
    worst_part = max(joint_analysis, key=joint_analysis.get)
    
    feedback['strengths'].append(f"{best_part}动作准确")
    feedback['weaknesses'].append(f"{worst_part}动作偏差较大")
    
    # 具体建议
    if worst_part == 'left_hand':
        feedback['suggestions'].append("注意左手手势，参考标准动作调整手指弯曲度")
    elif worst_part == 'right_hand':
        feedback['suggestions'].append("注意右手手势，手腕角度需要调整")
    elif worst_part == 'body':
        feedback['suggestions'].append("注意身体姿势，肩膀和手臂位置需要调整")
    
    return feedback
```

## 4. 完整实现示例

### 4.1 评测系统主类

```python
import cv2
import mediapipe as mp
import numpy as np
from typing import Dict, List, Tuple

class SignLanguageEvaluator:
    """手语动作准确度评测系统"""
    
    def __init__(self, standard_db_path='./standard_db'):
        self.mp_holistic = mp.solutions.holistic
        self.holistic = self.mp_holistic.Holistic(
            min_detection_confidence=0.5,
            min_tracking_confidence=0.5
        )
        
        # 加载标准动作库
        self.standard_samples = self._load_standard_db(standard_db_path)
        
        # 评分权重
        self.weights = {
            'joint': 0.5,
            'temporal': 0.3,
            'speed': 0.2
        }
    
    def _load_standard_db(self, path):
        """加载标准动作数据库"""
        # 从文件加载标准特征
        # 格式：{semantic: [feature1, feature2, ...]}
        import json
        with open(f'{path}/metadata.json', 'r') as f:
            metadata = json.load(f)
        
        features = np.load(f'{path}/features.npy')
        
        standard_samples = {}
        for i, meta in enumerate(metadata):
            semantic = meta['semantic']
            if semantic not in standard_samples:
                standard_samples[semantic] = []
            standard_samples[semantic].append(features[i])
        
        return standard_samples
    
    def extract_features(self, video_path: str, target_frames: int = 30):
        """提取视频特征"""
        cap = cv2.VideoCapture(video_path)
        frames = []
        
        while cap.isOpened():
            ret, frame = cap.read()
            if not ret:
                break
            
            image = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            results = self.holistic.process(image)
            
            feature = self._extract_frame_features(results)
            frames.append(feature)
        
        cap.release()
        
        if len(frames) == 0:
            return None
        
        # 时序归一化
        frames = np.array(frames)
        normalized = self._temporal_normalize(frames, target_frames)
        return normalized
    
    def _extract_frame_features(self, results):
        """提取单帧特征"""
        features = []
        
        # 左手
        if results.left_hand_landmarks:
            for lm in results.left_hand_landmarks.landmark:
                features.extend([lm.x, lm.y, lm.z])
        else:
            features.extend([0] * 63)
        
        # 右手
        if results.right_hand_landmarks:
            for lm in results.right_hand_landmarks.landmark:
                features.extend([lm.x, lm.y, lm.z])
        else:
            features.extend([0] * 63)
        
        # 身体
        if results.pose_landmarks:
            for lm in results.pose_landmarks.landmark:
                features.extend([lm.x, lm.y, lm.z])
        else:
            features.extend([0] * 99)
        
        # 面部（简化）
        if results.face_landmarks:
            face_features = []
            for lm in results.face_landmarks.landmark:
                face_features.extend([lm.x, lm.y, lm.z])
            face_mean = np.mean(face_features)
            features.extend([face_mean] * 100)
        else:
            features.extend([0] * 100)
        
        return np.array(features, dtype=np.float32)
    
    def _temporal_normalize(self, frames, target_frames):
        """时序归一化"""
        from scipy.interpolate import interp1d
        
        if len(frames) >= target_frames:
            indices = np.linspace(0, len(frames)-1, target_frames, dtype=int)
            return frames[indices]
        else:
            original_frames = len(frames)
            f = interp1d(range(original_frames), frames, axis=0)
            return f(np.linspace(0, original_frames-1, target_frames))
    
    def evaluate(self, user_video: str, semantic: str) -> Dict:
        """
        评测用户动作
        
        参数：
        - user_video: 用户视频路径
        - semantic: 手语语义（如"你好"）
        
        返回：评测结果字典
        """
        # 1. 提取用户特征
        user_features = self.extract_features(user_video)
        if user_features is None:
            return {'error': '无法提取用户特征'}
        
        # 2. 获取标准样本
        if semantic not in self.standard_samples:
            return {'error': f'未找到手语"{semantic}"的标准样本'}
        
        standard_list = self.standard_samples[semantic]
        
        # 3. 对每个标准样本计算相似度，取最佳
        best_score = 0
        best_standard = None
        
        for std_feat in standard_list:
            # 整体相似度
            similarity = self._cosine_similarity(user_features.flatten(), std_feat.flatten())
            if similarity > best_score:
                best_score = similarity
                best_standard = std_feat
        
        # 4. 逐关节差异分析
        joint_diff = self._joint_analysis(user_features, best_standard)
        
        # 5. 计算综合分数
        score = self._calculate_score(joint_diff)
        
        # 6. 生成反馈
        feedback = self._generate_feedback(joint_diff, score)
        
        return {
            'semantic': semantic,
            'score': score,
            'similarity': round(best_score, 3),
            'joint_analysis': joint_diff,
            'feedback': feedback
        }
    
    def _cosine_similarity(self, a, b):
        """余弦相似度"""
        return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))
    
    def _joint_analysis(self, user_feat, std_feat):
        """逐关节差异分析"""
        # 按部位计算差异
        user_hand_l = user_feat[:, :63].flatten()
        user_hand_r = user_feat[:, 63:126].flatten()
        user_body = user_feat[:, 126:225].flatten()
        
        std_hand_l = std_feat[:, :63].flatten()
        std_hand_r = std_feat[:, 63:126].flatten()
        std_body = std_feat[:, 126:225].flatten()
        
        # 归一化差异（0-1）
        def norm_diff(a, b):
            diff = np.linalg.norm(a - b) / (np.linalg.norm(a) + 1e-6)
            return min(diff, 1.0)
        
        return {
            'left_hand': norm_diff(user_hand_l, std_hand_l),
            'right_hand': norm_diff(user_hand_r, std_hand_r),
            'body': norm_diff(user_body, std_body),
            'overall': np.mean([
                norm_diff(user_hand_l, std_hand_l),
                norm_diff(user_hand_r, std_hand_r),
                norm_diff(user_body, std_body)
            ])
        }
    
    def _calculate_score(self, joint_diff):
        """计算分数"""
        joint_score = (1 - joint_diff['overall']) * 100
        
        # 简化：暂时只使用关节差异
        total_score = joint_score * self.weights['joint'] + 50 * (1 - self.weights['joint'])
        
        # 等级
        if total_score >= 90:
            grade = '优秀'
        elif total_score >= 80:
            grade = '良好'
        elif total_score >= 60:
            grade = '及格'
        else:
            grade = '不及格'
        
        return {
            'total': round(total_score, 1),
            'grade': grade
        }
    
    def _generate_feedback(self, joint_diff, score):
        """生成反馈"""
        feedback = {
            'score': score['total'],
            'grade': score['grade'],
            'strengths': [],
            'weaknesses': [],
            'suggestions': []
        }
        
        # 找出最好和最差的部位
        parts = ['left_hand', 'right_hand', 'body']
        part_names = {'left_hand': '左手', 'right_hand': '右手', 'body': '身体'}
        
        best_part = min(joint_diff, key=joint_diff.get)
        worst_part = max(joint_diff, key=joint_diff.get)
        
        if best_part != 'overall':
            feedback['strengths'].append(f"{part_names[best_part]}动作较为准确")
        
        if worst_part != 'overall':
            feedback['weaknesses'].append(f"{part_names[worst_part]}动作偏差较大")
            
            if worst_part == 'left_hand':
                feedback['suggestions'].append("建议：注意左手手势，手指弯曲度和手腕角度需调整")
            elif worst_part == 'right_hand':
                feedback['suggestions'].append("建议：注意右手手势，参考标准动作调整")
            elif worst_part == 'body':
                feedback['suggestions'].append("建议：注意身体姿势，肩膀和手臂位置需调整")
        
        return feedback
    
    def close(self):
        self.holistic.close()


# ============== 使用示例 ==============
if __name__ == '__main__':
    # 初始化评测系统
    evaluator = SignLanguageEvaluator(standard_db_path='./standard_db')
    
    # 评测用户动作
    result = evaluator.evaluate(
        user_video='./user_test.mp4',
        semantic='你好'
    )
    
    # 输出结果
    print(f"手语：{result['semantic']}")
    print(f"准确度：{result['score']['total']}分 ({result['score']['grade']})")
    print(f"整体相似度：{result['similarity']}")
    print(f"\n详细分析:")
    for part, diff in result['joint_analysis'].items():
        if part != 'overall':
            print(f"  {part}: 差异 {diff:.3f}")
    
    print(f"\n反馈:")
    for s in result['feedback']['strengths']:
        print(f"  ✓ {s}")
    for w in result['feedback']['weaknesses']:
        print(f"  ✗ {w}")
    for sug in result['feedback']['suggestions']:
        print(f"  → {sug}")
    
    evaluator.close()
```

## 5. 标准动作库建设

### 5.1 录制规范

| 要求 | 说明 |
